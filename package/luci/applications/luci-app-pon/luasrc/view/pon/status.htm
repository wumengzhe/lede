" 2>/dev/null") if not f then return {} end local out = f:read("*a") or "" f:close() local kv = {} for line in out:gmatch("[^\n]+") do local k, v = line:match("^(\w+)=(.*)$") if k then kv[k] = v end end return kv end local st = ponctl("status") local states = { [0] = "No module", [1] = "O1 (initial)", [2] = "O2 (standby)", [3] = "O3 (serial number)", [4] = "O4 (ranging)", [5] = "O5 (operation)", } local modes = { [0] = "Auto", [1] = "XG-PON", [7] = "XGS-PON", } local state_name = states[tonumber(st.state)] or "unknown" local mode_name = modes[tonumber(st.mode)] or (st.mode or "?") local function fmt10(v) v = tonumber(v) or 0 return string.format("%.2f", v / 100) end local uci = require "luci.model.uci".cursor() local pcfg = uci:get("pon", "config") or {} local cfg_serial = pcfg.serial_no or "?" local cfg_loid = pcfg.lo_id or "-" local cfg_pass = (pcfg.password and #pcfg.password > 0) and "set" or "-" -%> <%+header%>

<%:Status%>
<%:Item%> <%:Value%>

<%:Serial number%> <%=cfg_serial%>
<%:LOID%> <%=cfg_loid%>
<%:Password%> <%=cfg_pass%>
<%:PON mode%> <%=mode_name%> (<%=st.mode or "?"%>)
<%:Laser%> <%=(st.laser == "1") and "on" or "off"%>
<%:RX power%> <%=fmt10(st.rx_power)%> dBm
<%:TX power%> <%=fmt10(st.tx_power)%> dBm
<%:Bias current%> <%=fmt10(st.bias)%> mA
<%:Temperature%> <%=fmt10(st.temperature)%> °C
<%:Voltage%> <%=(tonumber(st.voltage) or 0)%> mV
<%:FEC%> <%=(st.fec == "1") and "on" or "off"%>
<%:LOS%> <%=(st.los == "1") and "yes" or "no"%>
<%:TX fault%> <%=(st.tx_fault == "1") and "yes" or "no"%>
