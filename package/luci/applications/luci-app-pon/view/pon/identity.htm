<% -- Read board MAC + PON serial from factory nvmem cells. -- bosa volume has calibration@0 (256 B used); ri volume has pon-serial@1a -- (12 B, 4 ASCII vendor + 8 hex) and macaddr@3e (6 B MAC). -- The simplest robust path on 6.18.x is sysfs binary reads via /bin/cat. local function read_file(path) local f = io.open(path, "rb") if not f then return nil end local d = f:read("*a") f:close() return d end local function find_nvmem_cell(prefix) -- /sys/bus/nvmem/devices/*/nvmem-style cells local p = io.popen("ls /sys/bus/nvmem/devices/ 2>/dev/null") local list = p:read("*a") or "" p:close() for d in list:gmatch("[^\n]+") do local stem = d:match("^([^.]+)") if stem and (stem == prefix or stem:find(prefix, 1, true)) then return "/sys/bus/nvmem/devices/" .. d end end return nil end local function mac_str(raw) if not raw or #raw < 6 then return "(unavailable)" end local b1, b2, b3, b4, b5, b6 = raw:byte(1), raw:byte(2), raw:byte(3), raw:byte(4), raw:byte(5), raw:byte(6) return string.format("%02x:%02x:%02x:%02x:%02x:%02x", b1, b2, b3, b4, b5, b6) end local function ascii_clean(raw) if not raw then return "" end local s = "" for i = 1, #raw do local c = raw:sub(i, i) if c >= " " and c <= "~" then s = s .. c end end return s:gsub("^%s+", ""):gsub("%s+$", "") end local mac_path = find_nvmem_cell("3e") local sn_path = find_nvmem_cell("1a") local mac_raw = read_file(mac_path) local sn_raw = read_file(sn_path) -- fallback: try ETH0 mac + NVMEM alias path if not mac_raw then mac_path = "/sys/class/net/eth0/address" local f = io.open(mac_path, "r") if f then mac_raw = f:read("*l") or ""; f:close() end end %> <%+footer%>
